Future<void> updateSalary() async {
  final supabase = Supabase.instance.client;
  final now = DateTime.now();
  final currentMonth = DateFormat('yyyy-MM').format(now);
  final firstDayOfMonth = DateTime(now.year, now.month, 1);
  final lastDayOfMonth = DateTime(now.year, now.month + 1, 0); // Correct last day
  final today = now.day;

  if (today < 1 || today > 29) {
    return; // Only run between the 25th and 29th
  }

  try {
    // Fetch all active teachers
    final teachers = await supabase
        .from('teacher')
        .select('serial_no, teacher_id, name, salary, resign_status');

    for (var teacher in teachers) {
      if (teacher['resign_status'] == true) continue; // Skip resigned teachers

      // Check if salary record exists for this teacher for the current month
      final existingRecord = await supabase
          .from('salary')
          .select('serial_no')
          .eq('serial_no', teacher['serial_no'])
          .gte('created_at', firstDayOfMonth.toIso8601String())
          .lte('created_at', lastDayOfMonth.toIso8601String())
          .maybeSingle();

      if (existingRecord == null) {
        // Insert new salary record if it doesn't exist
        await supabase.from('salary').insert({
          'serial_no': teacher['serial_no'],
          'teacher_id': teacher['teacher_id'],
          'name': teacher['name'],
          'salary': teacher['salary'],
          'amount_paid': 0,
          'status': 'pending',
          'resign_status': teacher['resign_status'],
          'created_at': now.toIso8601String(),
          'updated_at': now.toIso8601String(),
        });
      }
    }
  } catch (e) {
    print('Error updating salary records: $e');
  }
}
